// URL gerada pelo Google Apps Script da sua planilha const URL_WEBHOOK_PLANILHA = "https://script.google.com/macros/s/SUA_URL_AQUI/exec"; function enviarRelatoPassageiro() { const dados = { id_ativo: document.getElementById('campo-id').value, nome_ativo: document.getElementById('campo-nome').value, tipo_outorga: document.getElementById('campo-outorga').value, avaliacao: document.getElementById('campo-avaliacao').value, problema: document.getElementById('campo-problema').value, comentario: document.getElementById('campo-comentario').value, lat: document.getElementById('campo-lat').value, lng: document.getElementById('campo-lng').value, precisao: document.getElementById('campo-precisao').value }; document.getElementById('btn-enviar').innerText = "Enviando..."; document.getElementById('btn-enviar').disabled = true; fetch(URL_WEBHOOK_PLANILHA, { method: "POST", mode: "no-cors", headers: { "Content-Type": "application/json" }, body: JSON.stringify(dados) }) .then(() => { // Tela de confirmação amigável para o cidadão document.getElementById('bloco-formulario').style.display = 'none'; document.getElementById('mensagem-sucesso').style.display = 'block'; }) .catch(err => { alert("Falha no envio. Tente novamente."); document.getElementById('btn-enviar').innerText = "Enviar Avaliação"; document.getElementById('btn-enviar').disabled = false; }); }
